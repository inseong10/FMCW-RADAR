"""Additional clock/identity audit after the two standard E14 analyses.

Run from the repository root, after compare_dma_frame_board.py --clock-hz 100000000
and dma_frame/analyze_profile.py --clock-hz 100000000 on the collected raw files.
Large spectra remain in build/; this driver archives compact evidence and hashes.
"""
import argparse
import hashlib
import json
from pathlib import Path
import re
import shutil
import sys

ROOT = Path.cwd()
sys.path.insert(0, str(ROOT/'benchmarks'))
from compare_dma_frame_board import irq_counts


def require(condition, message):
    if not condition:
        raise ValueError(message)


def digest(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def manifest(path):
    result = {}
    for line in Path(path).read_text().splitlines():
        sha, name = line.split(maxsplit=1)
        name = name.lstrip('*')
        require(name not in result and re.fullmatch('[0-9a-f]{64}', sha), 'Invalid manifest')
        result[name] = sha
    return result


def clock(path):
    text = Path(path).read_text()
    boot = re.findall(r'^([0-9a-f]{8}(?:-[0-9a-f]{4}){3}-[0-9a-f]{12})$', text, re.M)
    require(len(boot) == 1, 'Missing unique boot ID')
    regs = {int(a, 16): int(b, 16) for a, b in re.findall(r'^(0x[0-9A-Fa-f]+): (0x[0-9A-Fa-f]+)$', text, re.M)}
    require(set(regs) == {0xf8000100,0xf8000104,0xf8000108,0xf800010c,0xf8000120}, 'Incomplete CPU registers')
    octets = re.findall(r'^0000000\s+([0-7]{3})\s+([0-7]{3})\s+([0-7]{3})\s+([0-7]{3})\s*$', text, re.M)
    require(len(octets) == 1, 'Missing reference-clock DT bytes')
    ref = int.from_bytes(bytes(int(v, 8) for v in octets[0]), 'big')
    pll, control = regs[0xf8000100], regs[0xf8000120]
    require(((control >> 4) & 3) <= 1 and control & (1 << 24), 'CPU not using enabled ARM PLL clock')
    require(not (pll & 0x13) and regs[0xf800010c] & 1, 'ARM PLL bypass/reset/powerdown/unlocked')
    multiplier, divisor = (pll >> 12) & 127, (control >> 8) & 63
    require(multiplier > 0 and divisor > 0, 'Invalid PLL/divider')
    hz = ref * multiplier // divisor
    require(hz == 666666660, 'Expected selected CPU configuration about 667MHz')
    return dict(boot_id=boot[0], registers={hex(k):hex(v) for k,v in regs.items()},
                ps_reference_dt_hz=ref, arm_pll_multiplier=multiplier, cpu_divisor=divisor,
                cpu_configuration_hz=hz, physical_frequency_counter_measurement=False)


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('results', type=Path)
    parser.add_argument('--source', type=Path)
    parser.add_argument('--source-tree', type=Path, required=True)
    parser.add_argument('--bundle', type=Path, required=True)
    args = parser.parse_args()
    data = args.results
    report = ROOT/'benchmarks/reports/2026-11-26-e14-clock'
    compact = report/'raw_compact'
    require(not compact.exists(), 'Preserve existing compact evidence')
    derived = {'comparison_console.json','profile_console.json','paired/comparison.json','profile/pl.profile.summary.json'}
    raw = {p.relative_to(data).as_posix():digest(p) for p in sorted(data.rglob('*'))
           if p.is_file() and p.relative_to(data).as_posix() not in derived}
    require(len(raw) == 83, 'Unexpected raw-file inventory')
    if args.source:
        source_raw = {p.relative_to(args.source).as_posix():digest(p) for p in sorted(args.source.rglob('*')) if p.is_file()}
        require(source_raw == raw, 'Shared raw collection differs from local mirror')

    comparison = json.loads((data/'paired/comparison.json').read_text())
    profile = json.loads((data/'profile/pl.profile.summary.json').read_text())
    require(comparison['status'] == 'E14_FRAME_PAIRED_SAME_BOARD_VALIDATED', 'Paired analysis did not pass')
    require(profile['frames'] == 20 and profile['status'] == 'E14_FRAME_DMA_DIAGNOSTIC_ONLY', 'Profile analysis did not pass')
    before, after = clock(data/'clock_before.txt'), clock(data/'clock_after.txt')
    require(before == after, 'CPU clock or boot changed during measurement')
    boot = before['boot_id']
    for phase in ('paired','profile'):
        require(boot in (data/phase/'environment.txt').read_text(), 'Boot ID mismatch')
    require(comparison['dma_irq_deltas'] == {'67':[330,0], '68':[330,0]}, 'Paired DMA IRQ ownership/count differs')
    profile_irq = [irq_counts(data/'profile'/('interrupts_'+phase+'.txt')) for phase in ('before','after')]
    irq_delta = {str(irq):[profile_irq[1][irq][cpu]-profile_irq[0][irq][cpu] for cpu in (0,1)] for irq in profile_irq[0]}
    require(irq_delta == {'67':[22,0], '68':[22,0]}, 'Profile DMA IRQ ownership/count differs')

    expected = manifest(args.bundle/'artifacts.sha256')
    require(len(expected) == 30, 'Unexpected selected-bundle inventory')
    for name, sha in expected.items():
        require(digest(args.bundle/name) == sha, 'Selected host bundle changed: '+name)
    require(comparison['dma_artifacts'] == expected, 'Measured artifact identities differ from selected bundle')
    require(manifest(data/'profile/artifacts.sha256') == expected, 'Profile/paired bundle differs')
    log = (data/'load_board.console.log').read_text()
    require(all(name+': OK' in log for name in expected), 'Board load did not verify every artifact')
    for marker in ('DMA_FRAME_CLIENT_PASS','SELFTEST_PASS backend=PL_DMA_FRAME frames=4 bins=65536','DMA_E14_FRAME_SELFTEST_READY'):
        require(marker in log, 'Missing on-board load/selftest marker')
    metadata = json.loads((args.bundle/'fft128_e14_frame.json').read_text())
    require(metadata['clock_hz'] == 100000000, 'DT overlay clock differs')
    require(json.loads((data/'paired/dma_config.json').read_text())['clock_hz'] == 100000000, 'Measurement clock differs')
    verified_sources = {}
    for kind in ('pl','host','module'):
        entries = manifest(data/'paired'/(kind+'_source.sha256'))
        for name, sha in entries.items():
            if kind == 'pl':
                prefix = '/home/user/fmcw_share/FMCW-FFT-test/'
                require(name.startswith(prefix), 'Unexpected PL source path')
                path = args.source_tree/name[len(prefix):]
            elif kind == 'host': path = args.source_tree/name
            else: path = args.source_tree/'src/PS/fft128/dma_frame'/name
            require(digest(path) == sha, 'Compiled source differs from shared source: '+name)
        verified_sources[kind] = entries

    old = json.loads((ROOT/'benchmarks/reports/2026-11-21-e14-frame-board/comparison.json').read_text())
    old_pl = old['results']['pl']['timing']['median_ms']
    pl = comparison['results']['pl']['timing']['median_ms']
    summary = dict(status='E14_100MHZ_BOARD_MEASUREMENTS_INDEPENDENTLY_VALIDATED',
                   raw_files=83, raw_bytes=sum((data/name).stat().st_size for name in raw),
                   clock_before=before, clock_after=after, cpu_clock_unchanged=True,
                   pl_clock_hz=100000000, selected_clock_is_maximum_claim=False,
                   paired_samples_per_variant={k:v['timing']['count'] for k,v in comparison['results'].items()},
                   paired_median_ms={k:v['timing']['median_ms'] for k,v in comparison['results'].items()},
                   double_over_pl=comparison['double_over_pl'], float_neon_over_pl=comparison['float_neon_over_pl'],
                   neon_to_pl_latency_reduction_percent=100*(1-1/comparison['float_neon_over_pl']),
                   paired_dma_irq_delta=comparison['dma_irq_deltas'], profile_dma_irq_delta=irq_delta,
                   profile_mean_ms=profile['mean_ms'], profile_wall_median_ms=profile['wall_median_ms'],
                   observed_old50_over_new100_pl=old_pl/pl,
                   observed_old50_to_new100_pl_reduction_percent=100*(1-pl/old_pl),
                   historical_comparison_limitation='Different session/boot; historical CPU clock not captured. Not an isolated paired clock-only comparison.',
                   deployed_bundle_artifact_sha256=expected, verified_compiled_source_sha256=verified_sources,
                   scope=comparison['scope'], limitation=comparison['limitation'],
                   stream_cfar_100mhz_verified=False)
    require(all(v == 300 for v in summary['paired_samples_per_variant'].values()), 'Missing paired repetitions')

    compact.mkdir()
    for name in raw:
        source = data/name
        if source.suffix == '.csv' and not name.endswith(('.timings.csv','.profile.csv')):
            continue
        dest = compact/name; dest.parent.mkdir(parents=True, exist_ok=True); shutil.copy2(source, dest)
    for name, value in [('board_measurements.json',summary),('raw_artifact_sha256.json',raw),
                        ('comparison.json',comparison),('profile_summary.json',profile)]:
        (report/name).write_text(json.dumps(value,indent=2)+'\n',encoding='utf-8')
    shutil.copy2(args.bundle/'artifacts.sha256',report/'bundle_artifacts.sha256')
    shutil.copy2(Path(__file__),report/'measurement_validation_driver.py.txt')
    for name in ('benchmarks/compare_dma_frame_board.py','src/PS/fft128/dma_frame/analyze_profile.py'):
        shutil.copy2(ROOT/name,report/(Path(name).name+'.txt'))
    code = {name:digest(ROOT/name) for name in ('benchmarks/compare_dma_frame_board.py',
             'src/PS/fft128/dma_frame/analyze_profile.py','benchmarks/compare_board.py',
             'benchmarks/ps_precision/prepare_reference.py','src/PS/fft128/receiver.py')}
    code['measurement_validation_driver.py.txt'] = digest(Path(__file__))
    (report/'measurement_analysis_sha256.json').write_text(json.dumps(code,indent=2)+'\n')
    print(json.dumps({k:summary[k] for k in ('status','raw_files','paired_median_ms','double_over_pl','float_neon_over_pl',
                     'profile_mean_ms','observed_old50_over_new100_pl','observed_old50_to_new100_pl_reduction_percent')},indent=2))


if __name__ == '__main__':
    main()
