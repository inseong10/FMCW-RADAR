def same(x,y,path=''):
    if isinstance(x,dict):
        assert x.keys()==y.keys(),path
        for k in x: same(x[k],y[k],path+'/'+k)
    elif isinstance(x,list):
        assert len(x)==len(y),path
        for i,(u,v) in enumerate(zip(x,y)):same(u,v,path+'/'+str(i))
    elif isinstance(x,float):
        assert math.isclose(x,y,rel_tol=1e-13,abs_tol=2e-12),(path,x,y)
        if x!=y:roundoff.append(dict(field=path,windows=x,ubuntu=y))
    elif path=='/limitation':
        old='Same recorded warm-cache frame, CPU0, one ioctl/axis, coherent HP0 bounce buffers; four unchanged cores with group barrier, no next-group prefetch or external ping-pong or continuous-stream throughput claim'
        new='Same recorded warm-cache frame, CPU0, one ioctl/axis, coherent HP0 bounce buffers; four unchanged cores with independent per-lane input/output FIFOs, next-vector prefetch and no group barrier; shared ordered DMA; no continuous-stream throughput claim'
        assert x==new and y==old,(path,x,y)
    else:assert x==y,(path,x,y)
